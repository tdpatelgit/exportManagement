{# Shared goods-row renderer for the invoice sheet and the annexure sheet. Groups consecutive items under a category heading row (first line of the name, HSN code in its own column), like the original Excel documents. Every item line gets a running number in the Sr No column, numbering continues across the annexure, in both display modes: - mode 'index' : items are grouped by name/HSN only. - mode 'surface' : items are additionally split by their surface finish, and the group's SURFACE (GLOSSY / MATT / ...) prints in the Sr No column of the heading row - so a heading row always prints in this mode, even for a lone item, to keep the surface visible without covering up its line number below. An item whose name has extra lines below the first prints those lines as its description (the first line lives in the heading); a single-line item prints just its quantities under the heading. #} {% macro group_key(item, mode) -%} {{- (item.product_name or '').split('\n')[0]|trim|upper }}|{{ item.hsn_code or '' }}{% if mode == 'surface' %}|{{ (item.surface or '')|trim|upper }}{% endif -%} {%- endmacro %} {# How many items print on the invoice page before the rest spill onto the ANNEXURE - decided by the height the goods rows will take, not a fixed item count, so a list of short single-line items fills the page instead of breaking after a handful. Walks the same grouping as rows() below and estimates each printed row in px (8px bold text: a 14px row, +10px per wrapped line). GOODS_BUDGET_PX is the room the invoice page leaves for goods rows; when a split is needed, SPLIT_RESERVE_PX pays for the "(CONTINUE TO ANNEXURE)" row and the Annexure charge line it adds. Returns the index as text - pipe it through |int. Used by both _sheet.html and _annexure.html, so the two can't disagree on where the split falls. #} {# Measured on a real render: with the usual 3-4 line addresses the page has ~380px for goods rows; kept under that for longer headers and the 30px filler floor. #} {% set GOODS_BUDGET_PX = 100 %} {% set SPLIT_RESERVE_PX = 34 %} {% set DESC_CHARS_PER_LINE = 52 %} {% macro _cut(items, mode, budget) -%} {%- set st = namespace(key=none, first=true, used=0, cut=items|length) -%} {%- for item in items -%} {%- set key = group_key(item, mode) -%} {%- set next_key = group_key(items[loop.index], mode) if loop.index < items|length else none -%} {%- set name = (item.product_name or '') -%} {%- set inline = key != st.key and next_key != key and '\n' not in name and mode != 'surface' -%} {%- set c = namespace(px=0) -%} {%- if key != st.key -%} {%- set st.key = key -%} {%- if not inline %}{% set c.px = c.px + 14 %}{% endif -%} {%- endif -%} {%- set lines = [name.split('\n')[0]] if inline else name.split('\n')[1:] -%} {%- set ln = namespace(n=0) -%} {%- for line in (lines or ['']) %}{% set ln.n = ln.n + ((line|length + DESC_CHARS_PER_LINE - 1) // DESC_CHARS_PER_LINE or 1) %}{% endfor -%} {%- set c.px = c.px + 4 + 10 * ln.n -%} {%- if st.cut == items|length and st.used + c.px > budget %}{% set st.cut = [loop.index0, 1]|max %}{% endif -%} {%- set st.used = st.used + c.px -%} {%- set st.first = false -%} {%- endfor -%} {{- st.cut -}} {%- endmacro %} {% macro annex_start(items, mode) -%} {%- set n = _cut(items, mode, GOODS_BUDGET_PX)|int -%} {%- if n < items|length %}{% set n = _cut(items, mode, GOODS_BUDGET_PX - SPLIT_RESERVE_PX)|int %}{% endif -%} {{- n -}} {%- endmacro %} {% macro rows(items, mode, sr_start=0, currency_prefix='$') %} {% set gs = namespace(sr=sr_start, key=none, first=true) %} {% for item in items %} {% set key = group_key(item, mode) %} {% set is_new_group = key != gs.key %} {# A group with exactly one item and a single-line name doesn't need its own heading row above a near-empty numbered row - that split only earns its keep when a heading actually has multiple variant lines under it (surface finishes, or a multi-line product name). Detected by peeking at the next item, since group membership can't be known until the group closes. #} {% set next_key = group_key(items[loop.index], mode) if loop.index < items|length else none %} {% set is_solo = is_new_group and next_key != key and '\n' not in (item.product_name or '') %} {# In surface mode the heading row is the only place the surface name prints, so it always prints - even for a solo item - rather than borrowing the item row's own Sr No cell (which would blank out its number). In index mode a solo item still skips the heading, inlining its name into the numbered row as before. #} {% set inline_into_row = is_solo and mode != 'surface' %} {% if is_new_group %} {% set gs.key = key %} {% if not inline_into_row %} {% if mode == 'surface' %}{{ (item.surface or '')|upper }}{% endif %} {{ (item.product_name or '').split('\n')[0]|upper }} {{ item.hsn_code or '' }} {% endif %} {% endif %} {% set gs.first = false %} {% set gs.sr = gs.sr + 1 %} {% set rest = (item.product_name or '').split('\n')[1:] | join('\n') %} {{ gs.sr }} {%- if inline_into_row -%} {{ (item.product_name or '').split('\n')[0]|upper }} {%- else -%} {%- for line in rest.split('\n') -%} {{ line|upper }}{% if not loop.last %}
{% endif %} {%- endfor -%} {%- endif -%} {% if inline_into_row %}{{ item.hsn_code or '' }}{% endif %} {% if item.pallets %}{{ '{:,.0f}'.format(item.pallets) }} {{ {'PLT': 'PLTS', 'CTN': 'CTNS', 'BOX': 'BOXES', 'BAG': 'BAGS', 'SET': 'SETS'}.get(item.packing_unit or 'PLT', item.packing_unit) }}{% endif %} {% if item.quantity_boxes %}{{ '{:,.0f}'.format(item.quantity_boxes) }} {{ item.quantity_unit }}{% endif %} {{ '{:,.2f}'.format(item.quantity_value or 0) }} {{ item.unit }} {{ currency_prefix }}{{ '{:,.2f}'.format(item.price_usd or 0) }} / {{ item.unit }} {{ currency_prefix }} {{ '{:,.2f}'.format(item.total_usd or 0) }} {% endfor %} {% endmacro %}
